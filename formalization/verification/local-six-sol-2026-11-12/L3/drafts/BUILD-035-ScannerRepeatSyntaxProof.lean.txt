theorem rowSubst {alpha beta : Type} (T P : Program 2) (code : OTerm alpha)
 (ts : alpha → OTerm beta) :
 (row T P code).subst (fun a => emb (ts a)) = row T P (code.subst ts) := by
 unfold row matrix
 rw [allBlockSubst,liftedEmbedding,impSubst,conjunctionSubst]
 simp only [repeatSubst,graph2,programSubst,consTermSubst,liftSubst,
  Term.subst,Sum.elim_inr,emptySubst,sucSubst,zSubst]

theorem rowRename {alpha beta : Type} (T P : Program 2) (code : OTerm alpha)
 (rho : alpha → beta) :
 (row T P code).relabel rho = row T P (code.relabel rho) := by
 rw [formulaRelabelAsSubst]
 change (row T P code).subst (fun a => emb (.var (rho a))) = _
 rw [rowSubst]
 simp only [← termRelabelAsSubst]

theorem matrixWitness {alpha : Type} (T P : Program 2)
 (code initial out parameter : OTerm alpha) :
 (matrix T P code).subst
  (Sum.elim Term.var (fun i => emb (![initial,out,parameter] i))) =
 ((InternalParameterizedApplicationStepChecker.repeatGraph T out code initial parameter ⊓
   graph2 P (suc z) initial parameter).imp (graph2 P (suc z) out parameter)) := by
 have he : (Sum.elim Term.var (fun i => emb (![initial,out,parameter] i))) =
  (fun a => emb (Sum.elim Term.var ![initial,out,parameter] a)) := by
  funext a
  cases a <;> rfl
 unfold matrix
 rw [he,impSubst,conjunctionSubst]
 simp only [repeatSubst,graph2,programSubst,consTermSubst,witnessLift,
  Term.subst,Sum.elim_inr,emptySubst,sucSubst,zSubst]
 rfl

theorem rowAtCode {alpha : Type} (T P : Program 2) (code : OTerm alpha) :
 atCode (row T P (.var (.inr (0 : Fin 1)))) code = row T P code := by
 simp only [atCode,instantiate,bindEmbedding,rowSubst,varBind]

theorem rowAtIndex {alpha : Type} (T P : Program 2) (i : Fin 3) :
 atIndex (row T P (.var (.inr (0 : Fin 1))) : Formula (alpha ⊕ Fin 1)) i =
  row T P (.var (.inr i)) := by
 rw [atIndex,rowRename]
 rfl
