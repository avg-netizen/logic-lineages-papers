variable {body : Positive (Fin 1)} {alpha : Type} {Gamma : List (Formula alpha)}

theorem recoveredRename {beta : Type} (data : Data) (oldStep : Program 4)
    (source target step sourceOut targetOut certificate parameter : OTerm alpha) (j : alpha → beta) :
    (Recovered data oldStep source target step sourceOut targetOut certificate parameter).relabel j =
      Recovered data oldStep (source.relabel j) (target.relabel j) (step.relabel j)
        (sourceOut.relabel j) (targetOut.relabel j) (certificate.relabel j) (parameter.relabel j) := by
  simp only [Recovered,stepAccepted,edgeAccepted,InternalParameterizedCanonicalResultFold.foldGraph,
    conjunctionRename,graph2Rename,graph3Rename,graph4Rename,
    InternalOriginalStepGuard.one,sucRelabel,zRelabel]

/-- Rebuild the literal uniform quotient checker at the two actual fold
outputs. The accepted raw checker supplies both source-validity factors; the
source fold is filtered only after recovering that original validity. -/
theorem recoveredAt (data : Data) (oldStep : Program 4)
    (source target step sourceOut targetOut certificate parameter : OTerm alpha)
    (hstep : Nonempty (Derivation body Gamma (stepAccepted data oldStep source target step parameter)))
    (hs : Nonempty (Derivation body Gamma
      (InternalParameterizedCanonicalResultFold.foldGraph data sourceOut source parameter)))
    (ht : Nonempty (Derivation body Gamma
      (InternalParameterizedCanonicalResultFold.foldGraph data targetOut target parameter)))
    (hc : Nonempty (Derivation body Gamma (graph3 edgeCode certificate source target step))) :
    Nonempty (Derivation body Gamma
      (Recovered data oldStep source target step sourceOut targetOut certificate parameter)) := by
  obtain ⟨hvs,hvt,_⟩ := TypingProof.stepParts data oldStep source target step parameter hstep
  have hsource := ApplicationEncodedProof.sourceFromFold data sourceOut source parameter hvs hs
  have hedge := SupportProof.checkerAccept data oldStep sourceOut targetOut certificate
    source target step parameter (edgeSelectors certificate source target step hc)
    (andIntro hstep (andIntro hsource ht))
  exact andIntro hstep (andIntro hvs (andIntro hvt
    (andIntro hs (andIntro ht (andIntro hc hedge)))))

/-- Retain an already recovered original edgeCode, including the peeled
child certificate. Only its two actual canonical outputs are bound here. -/
theorem encodedCases (data : Data) (oldStep : Program 4)
    (source target step certificate parameter : OTerm alpha) (psi : Formula alpha)
    (hstep : Nonempty (Derivation body Gamma (stepAccepted data oldStep source target step parameter)))
    (hc : Nonempty (Derivation body Gamma (graph3 edgeCode certificate source target step)))
    (next : ∀ (beta : Type) (j : alpha → beta) (Delta : List (Formula beta)),
      (∀ h ∈ Gamma, h.relabel j ∈ Delta) →
      ∀ sourceOut targetOut : OTerm beta,
      Nonempty (Derivation body Delta
        (Recovered data oldStep (source.relabel j) (target.relabel j) (step.relabel j)
          sourceOut targetOut (certificate.relabel j) (parameter.relabel j))) →
      Nonempty (Derivation body Delta (psi.relabel j))) :
    Nonempty (Derivation body Gamma psi) := by
  let qs : Fin 2 → Program 4 :=
    ![.composition (InternalParameterizedCanonicalResultFold.foldN data) ![.projection 0,.projection 3],
      .composition (InternalParameterizedCanonicalResultFold.foldN data) ![.projection 1,.projection 3]]
  let xs : Fin 4 → OTerm alpha := Fin.cons source (Fin.cons target (Fin.cons step (Fin.cons parameter Fin.elim0)))
  apply queryCases qs xs psi
  intro beta j ys Delta hj hq
  have hstep' := renameInto hstep j Delta hj
  have hc' := renameInto hc j Delta hj
  simp only [stepAccepted,graph4Rename,InternalOriginalStepGuard.one,sucRelabel,zRelabel] at hstep'
  rw [graph3Rename] at hc'
  simp only [xs,consTermRename,emptyRename] at hq
  let xx : Fin 4 → OTerm beta := Fin.cons (source.relabel j)
    (Fin.cons (target.relabel j) (Fin.cons (step.relabel j) (Fin.cons (parameter.relabel j) Fin.elim0)))
  have hs := binaryRecover (InternalParameterizedCanonicalResultFold.foldN data)
    (.projection 0) (.projection 3) (ys 0) (source.relabel j) (parameter.relabel j) xx
    (hq 0) (projectionValue 0 xx) (projectionValue 3 xx)
  have ht := binaryRecover (InternalParameterizedCanonicalResultFold.foldN data)
    (.projection 1) (.projection 3) (ys 1) (target.relabel j) (parameter.relabel j) xx
    (hq 1) (projectionValue 1 xx) (projectionValue 3 xx)
  exact next beta j Delta hj (ys 0) (ys 1)
    (recoveredAt data oldStep (source.relabel j) (target.relabel j) (step.relabel j)
      (ys 0) (ys 1) (certificate.relabel j) (parameter.relabel j) hstep' hs ht hc')

/-- The source, target and original certificate are queried in one internal
scope, so every recovered graph uses the same composed input renaming. -/
theorem actualOriginalUniformStepQuotientElimination :
    ActualOriginalUniformStepQuotientElimination := by
  intro body data oldStep alpha Gamma source target step parameter psi hstep next
  let qs : Fin 3 → Program 4 :=
    ![.composition (InternalParameterizedCanonicalResultFold.foldN data) ![.projection 0,.projection 3],
      .composition (InternalParameterizedCanonicalResultFold.foldN data) ![.projection 1,.projection 3],
      .composition edgeCode ![.projection 0,.projection 1,.projection 2]]
  let xs : Fin 4 → OTerm alpha := Fin.cons source (Fin.cons target (Fin.cons step (Fin.cons parameter Fin.elim0)))
  apply queryCases qs xs psi
  intro beta j ys Delta hj hq
  have hstep' := renameInto hstep j Delta hj
  simp only [stepAccepted,graph4Rename,InternalOriginalStepGuard.one,sucRelabel,zRelabel] at hstep'
  simp only [xs,consTermRename,emptyRename] at hq
  let xx : Fin 4 → OTerm beta := Fin.cons (source.relabel j)
    (Fin.cons (target.relabel j) (Fin.cons (step.relabel j) (Fin.cons (parameter.relabel j) Fin.elim0)))
  have hs := binaryRecover (InternalParameterizedCanonicalResultFold.foldN data)
    (.projection 0) (.projection 3) (ys 0) (source.relabel j) (parameter.relabel j) xx
    (hq 0) (projectionValue 0 xx) (projectionValue 3 xx)
  have ht := binaryRecover (InternalParameterizedCanonicalResultFold.foldN data)
    (.projection 1) (.projection 3) (ys 1) (target.relabel j) (parameter.relabel j) xx
    (hq 1) (projectionValue 1 xx) (projectionValue 3 xx)
  have hc := compositionRecover edgeCode ![.projection 0,.projection 1,.projection 2]
    (ys 2) xx ![source.relabel j,target.relabel j,step.relabel j] (hq 2)
    (fun i ↦ Fin.cases (projectionValue 0 xx)
      (Fin.cases (projectionValue 1 xx)
        (Fin.cases (projectionValue 2 xx) (fun q ↦ Fin.elim0 q))) i)
  exact next beta j Delta hj (ys 0) (ys 1) (ys 2)
    (recoveredAt data oldStep (source.relabel j) (target.relabel j) (step.relabel j)
      (ys 0) (ys 1) (ys 2) (parameter.relabel j) hstep' hs ht hc)
