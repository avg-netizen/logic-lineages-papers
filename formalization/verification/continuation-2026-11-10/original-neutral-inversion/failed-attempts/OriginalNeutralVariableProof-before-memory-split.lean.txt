/-- Apply the accepted constructor and representative refinements to the
supplied original tuple, retaining that tuple instead of selecting a new one. -/
theorem variableInputsCalibrated (n : Nat) (arity : Fin n → Nat)
    (p : GlobalSourcePackage.Inputs) (hp : GlobalSourcePackage.InputsCalibrated n arity p) :
    GlobalSourcePackage.InputsCalibrated n arity (OriginalVariableRepresentative.refineInputs p) := by
  obtain ⟨hbase,hwindow,hsrc,hcomp,hstep,hvertex,hresult⟩ := hp
  obtain ⟨_,hnewbase⟩ :=
    OriginalVariableRepresentativeEvaluationProof.actualOriginalVariableRepresentativeRefinementEquality
      n arity p.base p.window hbase hwindow
  have hstepEq := OriginalConstructorLeafEvaluationProof.actualOriginalConstructorLeafRefinementEquality
    n arity p.oldStep hstep
  refine ⟨hnewbase,hwindow,hsrc,hcomp,?_,hvertex,hresult⟩
  intro v
  exact (hstepEq v).trans (hstep v)

/-- Neither original leaf refinement changes the endpoint. The representative
refinement changes only oldRepresentative; the endpoint's validity/member
fields do not inspect that field. This is equality of the literal ASTs. -/
theorem finalEndpoint (p : GlobalSourcePackage.Inputs) :
    OriginalNeutralLeaf.endpoint
      (OriginalVariableRepresentative.refineInputs (OriginalNeutralLeaf.refineInputs p)) =
      OriginalNeutralLeaf.endpoint p := rfl

/-- The once-refined constructor leaf retains the neutral leaf as an actual
acceptance factor. Its neutral law therefore holds at the final tuple's own
endpoint, original edge code, parameter and same-type bottom constructor. -/
theorem finalNeutralLaw (p : GlobalSourcePackage.Inputs) :
    OriginalNeutralLeaf.NeutralBottom
      (OriginalNeutralLeaf.endpoint
        (OriginalVariableRepresentative.refineInputs (OriginalNeutralLeaf.refineInputs p)))
      p.window
      (OriginalVariableRepresentative.refineInputs (OriginalNeutralLeaf.refineInputs p)).oldStep := by
  rw [finalEndpoint]
  intro body alpha Delta source target certificate parameter state typeCode
  let q := OriginalNeutralLeaf.refineInputs p
  let P := OriginalConstructorLeaf.accepted (OriginalConstructorLeaf.refined q.oldStep)
    source target certificate parameter ⊓
    (graph3 edgeCode state source target certificate ⊓
      (graph2 (BottomRepresentative.windowAt p.window) (suc z) source parameter ⊓
        (graph1 OriginalNeutralLeaf.neutralTest (suc z) source ⊓
          (graph2 (InternalParameterizedApplicationStepChecker.guardP (OriginalNeutralLeaf.endpoint p))
              z state parameter ⊓ graph1 InternalUnpair.left typeCode source))))
  have hp : Nonempty (Derivation body (P :: Delta) P) :=
    ⟨Derivation.hypothesis List.mem_cons_self⟩
  obtain ⟨hfactor⟩ := OriginalConstructorLeafInternalProof.originalFactor q.oldStep
    body alpha (P :: Delta) source target certificate parameter
  obtain ⟨haccepted⟩ := andLeft hp
  have hleaf : Nonempty (Derivation body (P :: Delta)
      (OriginalConstructorLeaf.accepted q.oldStep source target certificate parameter)) :=
    ⟨Derivation.implicationElimination hfactor haccepted⟩
  have h1 := andRight hp
  have h2 := andRight h1
  have h3 := andRight h2
  have h4 := andRight h3
  obtain ⟨h⟩ := OriginalNeutralLeafInternalProof.neutralBottom
    (OriginalNeutralLeaf.endpoint p) p.window p.oldStep
    source target certificate parameter state typeCode hleaf
    (andLeft h1) (andLeft h2) (andLeft h3) (andLeft h4) (andRight h4)
  exact ⟨Derivation.implicationIntroduction h⟩

/-- One input tuple precedes both refinements and every window. The final
source, computability and lift are shared by the full Package, original
neutral-leaf law, variable certificate inversion and variable leastness. -/
theorem actualOriginalNeutralVariableGlobalSelection :
    OriginalNeutralVariableTargets.ActualOriginalNeutralVariableGlobalSelection := by
  intro n arity
  obtain ⟨p,hp⟩ := GlobalSourcePackageProof.actualOriginalGlobalInputSelection n arity
  let q := OriginalNeutralLeaf.refineInputs p
  let r := OriginalVariableRepresentative.refineInputs q
  have hq : GlobalSourcePackage.InputsCalibrated n arity q :=
    OriginalNeutralLeafProof.refinedInputsCalibrated n arity p hp
  have hr : GlobalSourcePackage.InputsCalibrated n arity r := variableInputsCalibrated n arity q hq
  refine ⟨p,hp,hq,hr,?_,finalNeutralLaw p,?_⟩
  · exact (GlobalSourcePackageProof.actualOriginalGlobalSourcePackageConstruction n arity r hr).2.2
  · intro symbols S table hS
    exact ⟨OriginalVariableObjectCertificateProof.actualOriginalVariableObjectCertificateBottom
        n arity q hq symbols S table hS,
      OriginalVariableObjectLeastProof.actualOriginalVariableObjectLeast
        n arity q hq symbols S table hS⟩
