variable {body : Positive (Fin 1)} {α : Type} {Gamma : List (Formula α)}

theorem atRaw (data : Data) (window : Program 1)
    (arrow d e f a out parameter fm flag depth next mark b raw : OTerm α)
    (h : Nonempty (Derivation body Gamma (premises data window arrow d e f a out parameter fm flag depth next)))
    (hn : Nonempty (Derivation body Gamma
      (InternalParameterizedCanonicalResultFold.foldGraph (selectedData data window) b a parameter)))
    (hr : Nonempty (Derivation body Gamma (graph4 application raw d e f b))) :
    Nonempty (Derivation body Gamma
      ((BottomSpineMeasure.graph mark out).iff (graph2 InternalPairing.pairing mark flag next))) := by
  let selected := selectedData data window
  have hw := andLeft h
  have happ := andLeft (andRight h)
  have hfm := andLeft (andRight (andRight h))
  have hl := andLeft (andRight (andRight (andRight h)))
  have hdepth := andLeft (andRight (andRight (andRight (andRight h))))
  have hnext := andRight (andRight (andRight (andRight (andRight h))))
  have har := andLeft happ
  have hm := andLeft (andRight happ)
  have hf := andLeft (andRight (andRight happ))
  have ha := andLeft (andRight (andRight (andRight happ)))
  have hj := andRight (andRight (andRight (andRight happ)))
  have hff := InternalOriginalUniformDescentSelectionInternalProof.typedRecover
    selected.oldRepresentative (InternalOriginalUniformDescentSelection.validity selected) selected.member
    f arrow parameter hf
  have haf := InternalOriginalUniformDescentSelectionInternalProof.typedRecover
    selected.oldRepresentative (InternalOriginalUniformDescentSelection.validity selected) selected.member
    a d parameter ha
  have hb := andRight (certifiedAt selected b a d parameter
    (andLeft (andRight haf)) (andLeft (andRight (andRight haf))) hn)
  have hv := rawValidity selected arrow d e f b raw parameter har
    (andLeft (andRight (andRight hff))) (andLeft (andRight hb)) hm
    (andLeft (andRight hff)) (andLeft hb) hr
  have hout := combineRecover selected d e f b raw out parameter hr
    (applicationCombine selected d e f a out parameter b hn hj)
  have hwraw := BottomDescentStepProof.windowChange window (suc z) f raw parameter hw
  have hactive := conjunctionAccept (BottomRepresentative.windowAt window)
    (SpineApplicationIteration.selectedValidity data window)
    (Fin.cons raw (Fin.cons parameter Fin.elim0)) hwraw hv
  have hi := SpineApplicationSelectedProof.selectedIff data window out raw parameter mark hactive hout
  have hraw := SpineApplicationRawProof.rawEquation arrow d e f b raw fm flag depth next mark
    har (andLeft (andRight (andRight hff))) (andLeft (andRight hb)) hr hfm hl hdepth hnext
  exact chainIff _ _ _ hi hraw


end LogicLineages.OrdinalComparison.TerminationOrdering.ID1Completion.SourceComputabilityCompletion.SpineApplicationProof
