variable {body : Positive (Fin 1)} {α : Type} {Gamma : List (Formula α)}

/-- The window test reads only the second argument's context selector.
Changing the value therefore needs no equality or decoding premise. -/
theorem windowChange (window : Program 1) (out input next parameter : OTerm α)
    (h : Nonempty (Derivation body Gamma (graph2 (windowAt window) out input parameter))) :
    Nonempty (Derivation body Gamma (graph2 (windowAt window) out next parameter)) := by
  apply programCases InternalUnpair.left (Fin.cons parameter Fin.elim0) _
  intro β j Delta context hj hc
  rw [graph2Rename]
  have h' := renameInto h j Delta hj
  rw [graph2Rename] at h'
  have hc' : Nonempty (Derivation body Delta
      (graph1 InternalUnpair.left context (parameter.relabel j))) := by
    simpa only [graph1,consTermRename,emptyRename] using hc
  let xs : Fin 2 → OTerm β := ![input.relabel j,parameter.relabel j]
  let ys : Fin 2 → OTerm β := ![next.relabel j,parameter.relabel j]
  have hx := unaryValue InternalUnpair.left (.projection (1 : Fin 2)) context
    (parameter.relabel j) xs hc' (projectionValue 1 xs)
  have hy := unaryValue InternalUnpair.left (.projection (1 : Fin 2)) context
    (parameter.relabel j) ys hc' (projectionValue 1 ys)
  have hw := compositionRecover window (fun _ : Fin 1 => leftOf (.projection (1 : Fin 2)))
    (out.relabel j) xs ![context] h' (fun i => Fin.cases hx (fun k => Fin.elim0 k) i)
  exact compositionValue window (fun _ : Fin 1 => leftOf (.projection (1 : Fin 2)))
    (out.relabel j) ys ![context] hw (fun i => Fin.cases hy (fun k => Fin.elim0 k) i)

theorem activeFactors (data : Data) (window : Program 1) (input parameter : OTerm α)
    (h : Nonempty (Derivation body Gamma (activeGraph data window input parameter))) :
    Nonempty (Derivation body Gamma
      (graph2 (windowAt window) (suc z) input parameter ⊓
        validGraph (selectedValidity data window) input parameter)) := by
  exact conjunctionRecover (windowAt window) (selectedValidity data window) ![input,parameter] h

theorem activeNext (data : Data) (window : Program 1) (out input parameter : OTerm α)
    (ha : Nonempty (Derivation body Gamma (activeGraph data window input parameter)))
    (ht : Nonempty (Derivation body Gamma (graph2 (step data window) out input parameter))) :
    Nonempty (Derivation body Gamma (activeGraph data window out parameter)) := by
  have hf := activeFactors data window input parameter ha
  exact conjunctionAccept (windowAt window) (selectedValidity data window) ![out,parameter]
    (windowChange window (suc z) input out parameter (andLeft hf))
    (stepValid (representative data window) (selectedValidity data window) out input parameter ht (andRight hf))

/-- The actual choice either returns its input or the literal guarded old
representative output. Both branches preserve the existing bottom answer. -/
theorem forwardStep (data : Data) (window : Program 1) (out input parameter indicator : OTerm α)
    (ha : Nonempty (Derivation body Gamma (activeGraph data window input parameter)))
    (hb : Nonempty (Derivation body Gamma (bottomGraph indicator input)))
    (ht : Nonempty (Derivation body Gamma (graph2 (step data window) out input parameter))) :
    Nonempty (Derivation body Gamma (bottomGraph indicator out)) := by
  let R := representative data window
  let V := selectedValidity data window
  apply programCases R ![input,parameter] _
  intro β j Delta candidate hj hr
  rw [bottomRename]
  have hr' : Nonempty (Derivation body Delta
      (graph2 R candidate (input.relabel j) (parameter.relabel j))) := by
    simpa only [graph2,consTermRename,emptyRename] using hr
  have ha' := renameInto ha j Delta hj
  rw [activeRename] at ha'
  have hb' := renameInto hb j Delta hj
  rw [bottomRename] at hb'
  have ht' := renameInto ht j Delta hj
  rw [graph2Rename] at ht'
  let xs : Fin 2 → OTerm β := ![input.relabel j,parameter.relabel j]
  let ys : Fin 3 → OTerm β := ![candidate,input.relabel j,parameter.relabel j]
  have hc := compositionRecover (choose V) ![R,.projection 0,.projection 1]
    (out.relabel j) xs ys ht'
    (fun i => Fin.cases hr' (Fin.cases (projectionValue 0 xs)
      (Fin.cases (projectionValue 1 xs) (fun k => Fin.elim0 k))) i)
  let P := programGraph (guard V) (suc z) ys
  apply formulaCases P
  · have hp : Nonempty (Derivation body (P :: Delta) P) := ⟨Derivation.hypothesis List.mem_cons_self⟩
    have ho := successorBranchElim (guard V) (.projection (1 : Fin 3)) (.projection 0)
      (out.relabel j) z ys hp (withHyp hc P)
    obtain ⟨he⟩ := valueUnique (.projection (0 : Fin 3)) (out.relabel j) candidate ys ho (projectionValue 0 ys)
    obtain ⟨hi⟩ := SourceComputabilityCompletion.BottomRepresentativeInternalProof.preservation
      data window candidate (input.relabel j) (parameter.relabel j) (indicator.relabel j)
      (withHyp ha' P) (withHyp hr' P)
    obtain ⟨hb''⟩ := withHyp hb' P
    have hcandidate : Nonempty (Derivation body (P :: Delta) (bottomGraph (indicator.relabel j) candidate)) :=
      ⟨Derivation.implicationElimination (Derivation.conjunctionRight hi) hb''⟩
    exact headTransport bottomTest (indicator.relabel j) candidate (out.relabel j)
      Fin.elim0 ⟨symm he⟩ hcandidate
  · have hz := InternalBoundedDescentNormalizerStepSupportProof.conjunctionReject candidateLess
      (InternalAcceptance.conjunction (candidateValid V) sameLeft) ys
      (show Nonempty (Derivation body (P.not :: Delta) P.not) from ⟨Derivation.hypothesis List.mem_cons_self⟩)
    have ho := zeroBranchElim (guard V) (.projection (1 : Fin 3)) (.projection 0)
      (out.relabel j) ys hz (withHyp hc P.not)
    obtain ⟨he⟩ := valueUnique (.projection (1 : Fin 3)) (out.relabel j) (input.relabel j) ys ho (projectionValue 1 ys)
    exact headTransport bottomTest (indicator.relabel j) (input.relabel j) (out.relabel j)
      Fin.elim0 ⟨symm he⟩ (withHyp hb' P.not)
