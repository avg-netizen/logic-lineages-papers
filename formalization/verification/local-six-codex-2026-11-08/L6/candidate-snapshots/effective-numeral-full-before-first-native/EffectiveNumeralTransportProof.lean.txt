@[expose] public section
/-! Transport preserves the exact original numeric certificate, and dynamic
axiom families emit the original canonical singleton axiomLine certificate. -/
namespace LogicLineages.ComputabilityKnowledge.CertifiedSourceProvabilityCompletion.EffectiveNumeral.Transport
open GorbowSyntax GorbowProofChecker

private theorem axiomMono (ax bx : List Nat → Bool)
    (hmono : ∀ raw, ax raw = true → bx raw = true) (n : Nat) (raw : List Nat) (p : Payload)
    (h : GorbowTheoryChecker.axiomCheck ax n raw p = true) :
    GorbowTheoryChecker.axiomCheck bx n raw p = true := by
  simp only [GorbowTheoryChecker.axiomCheck, decide_eq_true_eq] at h ⊢
  rcases h with ⟨h0,h1,h2,h3,hv,ha,hs⟩
  exact ⟨h0,h1,h2,h3,hv,hmono _ ha,hs⟩

private theorem lineMono (ax bx : List Nat → Bool)
    (hmono : ∀ raw, ax raw = true → bx raw = true) (history : History) (line : Line)
    (h : GorbowTheoryChecker.lineCheck ax history line = true) :
    GorbowTheoryChecker.lineCheck bx history line = true := by
  have hp : GorbowSyntaxValidation.validFormula (context line) (conclusion line) = true ∧
      GorbowTheoryChecker.ruleCheck ax history line = true := by
    simpa only [GorbowTheoryChecker.lineCheck, Bool.and_eq_true] using h
  simp only [GorbowTheoryChecker.lineCheck, hp.1, Bool.true_and]
  by_cases h4 : rule line = 4
  · have ha : GorbowTheoryChecker.axiomCheck ax (context line) (conclusion line) (payload line) = true := by
      simpa only [GorbowTheoryChecker.ruleCheck,h4,if_true] using hp.2
    simpa only [GorbowTheoryChecker.ruleCheck,h4,if_true] using axiomMono ax bx hmono _ _ _ ha
  · simpa only [GorbowTheoryChecker.ruleCheck,h4,if_false] using hp.2

private theorem foldNone (ax : List Nat → Bool) (lines : List Line) :
    lines.foldl (GorbowTheoryChecker.step ax) none = none := by
  induction lines with
  | nil => rfl
  | cons line lines ih => simpa [List.foldl,GorbowTheoryChecker.step] using ih

private theorem runMono (ax bx : List Nat → Bool)
    (hmono : ∀ raw, ax raw = true → bx raw = true) (lines : List Line) :
    ∀ history out, GorbowTheoryChecker.run ax history lines = some out →
      GorbowTheoryChecker.run bx history lines = some out := by
  induction lines with
  | nil => intro history out h; exact h
  | cons line lines ih =>
    intro history out h
    by_cases hl : GorbowTheoryChecker.lineCheck ax history line = true
    · have hb := lineMono ax bx hmono history line hl
      have ht : GorbowTheoryChecker.run ax ((context line,conclusion line)::history) lines = some out := by
        simpa [GorbowTheoryChecker.run,List.foldl,GorbowTheoryChecker.step,hl] using h
      simpa [GorbowTheoryChecker.run,List.foldl,GorbowTheoryChecker.step,hb] using ih _ _ ht
    · simp [GorbowTheoryChecker.run,List.foldl,GorbowTheoryChecker.step,hl,foldNone] at h

private theorem verifiesMono (ax bx : List Nat → Bool)
    (hmono : ∀ raw, ax raw = true → bx raw = true) (n : Nat) (raw : List Nat) (lines : List Line)
    (h : GorbowTheoryChecker.verifies ax n raw lines = true) :
    GorbowTheoryChecker.verifies bx n raw lines = true := by
  cases hr : GorbowTheoryChecker.checkProof ax lines with
  | none => simp [GorbowTheoryChecker.verifies,hr] at h
  | some history =>
    have hb : GorbowTheoryChecker.checkProof bx lines = some history := runMono ax bx hmono lines [] history hr
    cases history <;> simpa [GorbowTheoryChecker.verifies,hr,hb] using h

theorem numericMonotonicity (ax bx : List Nat → Bool)
    (hmono : ∀ raw, ax raw = true → bx raw = true) (n formula certificate : Nat)
    (h : GorbowTheoryNumericCertificate.check ax n formula certificate = true) :
    GorbowTheoryNumericCertificate.check bx n formula certificate = true := by
  cases hf : GorbowSortedSyntaxCode.decodeFormulaRaw formula with
  | none => simp [GorbowTheoryNumericCertificate.check,hf] at h
  | some raw =>
    cases hc : GorbowNumericCertificate.decodeCertificate certificate with
    | none => simp [GorbowTheoryNumericCertificate.check,hf,hc] at h
    | some lines =>
      have hv : GorbowTheoryChecker.verifies ax n raw lines = true := by
        simpa [GorbowTheoryNumericCertificate.check,hf,hc] using h
      simpa [GorbowTheoryNumericCertificate.check,hf,hc] using verifiesMono ax bx hmono n raw lines hv

end LogicLineages.ComputabilityKnowledge.CertifiedSourceProvabilityCompletion.EffectiveNumeral.Transport

namespace LogicLineages.ComputabilityKnowledge.CertifiedSourceProvabilityCompletion.EffectiveNumeral.Family.HasCertificates
open GorbowSyntax GorbowTheoryProof
variable {alpha : Type} [Primcodable alpha]

theorem monotonic (ax bx : alpha → List Nat → Bool) {n : Nat} {phi : alpha → Formula n}
    (hm : ∀ x raw, ax x raw = true → bx x raw = true) (hc : HasCertificates ax phi) :
    HasCertificates bx phi := by
  obtain ⟨produce, hp, hv⟩ := hc
  exact ⟨produce, hp, fun x => Transport.numericMonotonicity (ax x) (bx x) (hm x)
    n _ _ (hv x)⟩

theorem axiomFamily (ax : alpha → List Nat → Bool) {sigma : alpha → Sentence}
    (hp : FormulaEffective sigma)
    (ha : ∀ x, ax x (GorbowFormulaPrefixCode.encode (sigma x)) = true) :
    HasCertificates ax sigma := by
  have hraw : Primrec (fun x => GorbowFormulaPrefixCode.encode (weakenClosed 0 (sigma x))) := by
    simpa only [GorbowClosedDeductionProof.closedSyntax.1] using hp
  have hargs := Primrec.list_cons.comp hp (Primrec.const [])
  have hline : Primrec (fun x => GorbowTheoryChecker.axiomLine 0 (sigma x)) :=
    (Primrec.const 0).pair (hraw.pair ((Primrec.const 4).pair
      ((Primrec.const 0).pair (hargs.pair ((Primrec.const 0).pair (Primrec.const 0))))))
  have hlist := Primrec.list_cons.comp hline (Primrec.const [])
  refine ⟨fun x => GorbowNumericCertificate.certificateNumber
      [GorbowTheoryChecker.axiomLine 0 (sigma x)], Primrec.encode.comp hlist, ?_⟩
  intro x
  have h := compiledCertificate (ax x)
    (Derivation.axiom (A := GorbowTheoryChecker.ProgramTheory (ax x)) (n := 0) (sigma x) (ha x))
  simpa only [GorbowTheoryChecker.proofLines, GorbowClosedDeductionProof.closedSyntax.1] using h

end LogicLineages.ComputabilityKnowledge.CertifiedSourceProvabilityCompletion.EffectiveNumeral.Family.HasCertificates
