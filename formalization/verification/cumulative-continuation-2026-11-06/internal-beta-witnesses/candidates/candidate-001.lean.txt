theorem actualBetaMatrix : InternalBetaWitnessTargets.ActualBetaMatrix := by
  intro α a b i r
  have h : OrdinaryArithmetic.betaFormula a b i r = OrdinaryArithmetic.existsFin (matrix a b i r) := by
    unfold OrdinaryArithmetic.betaFormula OrdinaryArithmetic.betaBody OrdinaryArithmetic.existsFin
    unfold matrix equations modulus
    congr 1
    simp only [BoundedFormula.relabel,BoundedFormula.mapTermRel,Term.equal,Term.bdEqual,
      BoundedFormula.inf,BoundedFormula.not,BoundedFormula.relabelAux,Term.relabel_relabel]
    rfl
  rw [h,actualOriginalBlockEmbedding]

theorem matrixSubst {α : Type} (a b i r : OTerm α) (ts : Fin 2 → OrdinaryArithmetic.Term α) :
    (arithmeticEmbedding.onFormula (matrix a b i r)).subst (Sum.elim Term.var (fun j => emb (ts j))) =
    arithmeticEmbedding.onFormula (equations a b i r (ts 0) (ts 1)) := by
  have he : (Sum.elim Term.var (fun j => emb (ts j))) =
      (fun x => emb (Sum.elim Term.var ts x)) := by
    funext x
    cases x <;> rfl
  rw [he,← embedSubst]
  have ht (t : OrdinaryArithmetic.Term α) : (t.relabel Sum.inl).subst (Sum.elim Term.var ts) = t := by
    rw [termRelabelSubst]
    exact termSubstVar t
  unfold matrix equations modulus
  congr 1
  change ((a.relabel Sum.inl).subst _).equal
      (add (mul (.var (.inr (0 : Fin 2))) (suc (mul (suc (i.relabel Sum.inl)) (b.relabel Sum.inl))))
        (r.relabel Sum.inl)).subst _ ⊓
    ((add (r.relabel Sum.inl) (suc (.var (.inr (1 : Fin 2))))).subst _).equal
      ((suc (mul (suc (i.relabel Sum.inl)) (b.relabel Sum.inl))).subst _) = _
  simp only [addSubst,mulSubst,sucSubst,ht,Term.subst,Sum.elim_inr]

theorem actualBetaIntroduction : InternalBetaWitnessTargets.ActualBetaIntroduction := by
  intro body α Gamma a b i r q d h
  rw [actualBetaMatrix]
  apply actualFiniteExistentialIntroduction body α 2 Gamma _ (fun j => emb (Fin.cases q (fun _ => d) j))
  rw [matrixSubst]
  exact h

theorem actualBetaElimination : InternalBetaWitnessTargets.ActualBetaElimination := by
  intro body α Gamma a b i r psi h e
  rw [actualBetaMatrix] at h
  exact actualFiniteExistentialElimination body α 2 Gamma _ psi h e

theorem actualBetaZero : InternalBetaWitnessTargets.ActualBetaZero := by
  intro body α b i
  apply actualBetaIntroduction body α [] z b i z z (mul (suc i) b)
  change Nonempty (Derivation body [] ((arithmeticEmbedding.onFormula
    (z.equal (add (mul z (modulus b i)) z))) ⊓
    arithmeticEmbedding.onFormula ((add z (suc (mul (suc i) b))).equal (modulus b i))))
  rw [onEquality,onEquality]
  apply Nonempty.intro
  apply Derivation.conjunctionIntroduction
  · exact symm (trans (addZero (mul z (modulus b i))) (multiplicationLeftZero (modulus b i)))
  · exact leftZero (suc (mul (suc i) b))
