theorem valuationPreservingSystem : SystemSourceTargets.ValuationPreservingSystem := by
  intro n s
  induction s with
  | nil =>
      intro ρ
      constructor
      · intro _
        exact ⟨ρ, (fun _ _ _ => rfl), (by simp [compileSystem, SourceAggregateCompletion.AllEqual])⟩
      · intro _ d hd
        exact False.elim (List.not_mem_nil hd)
  | cons d ds ih =>
      intro ρ
      let t := compileSystem ds
      let a := (SignedPair.compile d).pair.1.rename t.original
      let b := (SignedPair.compile d).pair.2.rename t.original
      change OrdinalSolves (d::ds) ρ ↔
        ∃ σ : AdmissibleSubstitution (.ground::.ground::t.context),
          Preserves (fun {_} x => shiftTwo t.context (t.original x)) ρ σ ∧
          SourceAggregateCompletion.AllEqual (valuation σ)
            ((left (alignedLeft a) (alignedRight b), right (alignedLeft a) (alignedRight b))::
              t.detectors.map (fun p =>
                (p.1.rename (shiftTwo t.context), p.2.rename (shiftTwo t.context))))
      constructor
      · intro hs
        have hd := hs d (List.mem_cons_self)
        have hds : OrdinalSolves ds ρ := fun e he => hs e (List.mem_cons_of_mem _ he)
        obtain ⟨θ,hθ,hall⟩ := (ih ρ).mp hds
        have hh : HeadEqual θ a b := by
          apply (headEqual_values θ a b).mpr
          dsimp only [a,b]
          rw [head_preserves t.original ρ θ hθ, head_preserves t.original ρ θ hθ]
          exact (compiled_zero_iff d ρ).mp hd
        obtain ⟨z,w,hvec⟩ := (FunctionalDetector.Proof.alignment t.context a b θ).mp hh
        let σ := extend θ z w
        refine ⟨σ, ?_, ?_⟩
        · intro τ x j
          change θ.vector (t.original x) j = ρ.vector x j
          exact hθ τ x j
        · apply (SourceAggregateCompletion.Proof.all_equal_cons _ _ _ _).mpr
          constructor
          · exact (FunctionalDetector.Proof.detector σ (alignedLeft a) (alignedRight b)).2.1.mpr hvec
          · exact (allEqual_preserves (shiftTwo t.context) θ σ (fun _ _ _ => rfl) _).mpr hall
      · rintro ⟨σ,hσ,hall⟩
        let θ := pullback (shiftTwo t.context) σ
        have hθ : Preserves t.original ρ θ := by
          intro τ x j
          exact hσ τ x j
        have hpairs := (SourceAggregateCompletion.Proof.all_equal_cons _ _ _ _).mp hall
        have hvec := (FunctionalDetector.Proof.detector σ (alignedLeft a) (alignedRight b)).2.1.mp hpairs.1
        have hh := aligned_reverse a b σ hvec
        have hd : SignedPair.meaning (SignedPair.inputs ρ) d = 0 := by
          apply (compiled_zero_iff d ρ).mpr
          dsimp only [a,b] at hh
          rw [head_preserves t.original ρ θ hθ, head_preserves t.original ρ θ hθ] at hh
          exact hh
        have htail := (allEqual_preserves (shiftTwo t.context) θ σ (fun _ _ _ => rfl) _).mp hpairs.2
        have hds := (ih ρ).mpr ⟨θ,hθ,htail⟩
        intro e he
        rcases List.mem_cons.mp he with he | he
        · subst e
          exact hd
        · exact hds e he

theorem systemEquality : SystemSourceTargets.SystemEquality := by
  intro n s
  constructor
  · rintro ⟨ρ,hρ⟩
    obtain ⟨σ,hσ,he⟩ := (valuationPreservingSystem n s ρ).mp hρ
    refine ⟨σ, ?_⟩
    exact (SourceAggregateCompletion.Proof.admissible_aggregate _ _ σ).2.1.mpr he
  · rintro ⟨σ,he⟩
    let ρ := pullback (compileSystem s).original σ
    refine ⟨ρ, (valuationPreservingSystem n s ρ).mpr ?_⟩
    exact ⟨σ, (fun _ _ _ => rfl),
      (SourceAggregateCompletion.Proof.admissible_aggregate _ _ σ).2.1.mp he⟩

theorem queryComplements : SystemSourceTargets.QueryComplements := by
  intro n s
  constructor
  · intro hu hs
    obtain ⟨ρ,he⟩ := (systemEquality n s).mp hs
    have hh := hu ρ
    omega
  · intro hn ρ
    have ha := SourceAggregateCompletion.Proof.admissible_aggregate _ (compileSystem s).detectors ρ
    have hne : rawScalar (valuation ρ) (strictSystemPair s).1 ≠
        rawScalar (valuation ρ) (strictSystemPair s).2 := by
      intro he
      exact hn ((systemEquality n s).mpr ⟨ρ,he⟩)
    have hle := ha.1
    change rawScalar _ (strictSystemPair s).2 ≤ rawScalar _ (strictSystemPair s).1 at hle
    omega

theorem weak_iff_strict {n : Nat} (s : System n)
    (ρ : AdmissibleSubstitution (compileSystem s).context) :
    rawScalar (valuation ρ) (weakSystemPair s).1 ≤ rawScalar (valuation ρ) (weakSystemPair s).2 ↔
    rawScalar (valuation ρ) (strictSystemPair s).2 < rawScalar (valuation ρ) (strictSystemPair s).1 := by
  unfold weakSystemPair ParametricLiteralWeakWrapper.constantApply ParametricLiteralWeakWrapper.identityApply
  rw [GroundEquivalence.Proof.constant_raw, GroundEquivalence.Proof.identity_raw,
    GroundEquivalence.Proof.g_le_iff]
  have ha := SourceAggregateCompletion.Proof.admissible_aggregate _ (compileSystem s).detectors ρ
  constructor
  · intro h
    omega
  · intro h
    have hn : ¬SourceAggregateCompletion.AllEqual (valuation ρ) (compileSystem s).detectors := by
      intro he
      have hh := ha.2.1.mpr he
      change rawScalar _ (strictSystemPair s).1 = rawScalar _ (strictSystemPair s).2 at hh
      omega
    have hgap := ha.2.2 hn
    change 2 ≤ rawScalar _ (strictSystemPair s).1 - rawScalar _ (strictSystemPair s).2 at hgap
    omega

theorem weakQueryComplements : SystemSourceTargets.WeakQueryComplements := by
  intro n s
  exact (forall_congr' (fun ρ => weak_iff_strict s ρ)).trans (queryComplements n s)

theorem polynomialQueryReduction : SystemSourceTargets.PolynomialQueryReduction := by
  intro n s
  have he := Proof.simultaneousActualPatterns n s
  constructor
  · constructor
    · intro hn z hz
      apply (queryComplements n (restrictSystem z s)).mpr
      intro h
      exact hn (he.mpr ⟨z,hz,h⟩)
    · intro hall hs
      obtain ⟨z,hz,ρ,hρ⟩ := he.mp hs
      exact ((queryComplements n _).mp (hall z hz)) ⟨ρ,hρ⟩
  · constructor
    · intro hn z hz
      apply (weakQueryComplements n (restrictSystem z s)).mpr
      intro h
      exact hn (he.mpr ⟨z,hz,h⟩)
    · intro hall hs
      obtain ⟨z,hz,ρ,hρ⟩ := he.mp hs
      exact ((weakQueryComplements n _).mp (hall z hz)) ⟨ρ,hρ⟩

end LogicLineages.OrdinalComparison.Source.FunctionalHardnessCompletion.SystemProof
