open Polynomial

theorem atTwoPower_identity (n : Nat) :
    (X-2)*signed (atTwoPower n) = (X : ZPolynomial)^n-C ((2^n : Nat) : Int) := by
  induction n with
  | zero => simp [atTwoPower,signed]
  | succ n ih =>
      have hc : signed (atTwoPower (n+1)) = X*signed (atTwoPower n)+C ((2^n : Nat) : Int) := by
        simp [atTwoPower,signed]
      rw [hc]
      calc
        (X-2)*(X*signed (atTwoPower n)+C ((2^n : Nat) : Int)) =
            X*((X-2)*signed (atTwoPower n))+(X-2)*C ((2^n : Nat) : Int) := by ring
        _ = X*((X : ZPolynomial)^n-C ((2^n : Nat) : Int))+
            (X-2)*C ((2^n : Nat) : Int) := by rw [ih]
        _ = (X : ZPolynomial)^(n+1)-C ((2^(n+1) : Nat) : Int) := by
          rw [pow_succ, Nat.cast_mul, Polynomial.C_mul]
          ring

theorem atTwoQuotient_add (p q : NPolynomial) :
    atTwoQuotient (p+q) = atTwoQuotient p+atTwoQuotient q := by
  apply Polynomial.sum_add_index
  · intro n
    simp
  · intro n a b
    simp [add_mul]

theorem atTwoQuotient_identity (p : NPolynomial) :
    (X-2)*signed (atTwoQuotient p) = signed p-C ((p.eval (2 : Nat) : Nat) : Int) := by
  induction p using Polynomial.induction_on' with
  | add p q hp hq =>
      rw [atTwoQuotient_add,signed_add,mul_add,hp,hq,signed_add,
        Polynomial.eval_add,Nat.cast_add,Polynomial.C_add]
      ring
  | monomial n c =>
      have hq : atTwoQuotient (monomial n c) = C c*atTwoPower n := by
        simp [atTwoQuotient,Polynomial.sum_monomial_index]
      rw [hq,signed_mul,signed_C]
      rw [←mul_assoc,mul_comm (X-2) (C (c : Int)),mul_assoc,atTwoPower_identity]
      simp only [signed,Polynomial.map_monomial,Polynomial.eval_monomial,Nat.cast_mul,Nat.cast_pow]
      change C (c : Int)*((X : ZPolynomial)^n-C ((2^n : Nat) : Int)) =
        monomial n (c : Int)-C ((c : Int)*(2 : Int)^n)
      rw [←Polynomial.C_mul_X_pow_eq_monomial,Polynomial.C_mul]
      ring

theorem doubleDifference_add (p q : NPolynomial) :
    doubleDifference (p+q) = doubleDifference p+doubleDifference q := by
  apply Polynomial.sum_add_index
  · intro n
    simp
  · intro n a b
    simp [add_mul]

theorem doubleDifference_identity (p : NPolynomial) :
    (X-1)*signed (doubleDifference p) = signed (p.comp (X^2))-signed p := by
  induction p using Polynomial.induction_on' with
  | add p q hp hq =>
      rw [doubleDifference_add,signed_add,mul_add,hp,hq,Polynomial.add_comp,signed_add,signed_add]
      ring
  | monomial n c =>
      have hq : doubleDifference (monomial n c) = C c*X^n*geometric n := by
        simp [doubleDifference,Polynomial.sum_monomial_index]
      rw [hq,signed_mul,signed_mul,signed_C,signed_X_pow]
      have hp : ((X : ZPolynomial)^2)^n = ((X : ZPolynomial)^n)^2 := by
        rw [←pow_mul,←pow_mul,Nat.mul_comm]
      calc
        (X-1)*(C (c : Int)*X^n*signed (geometric n)) =
            C (c : Int)*X^n*((X-1)*signed (geometric n)) := by ring
        _ = C (c : Int)*X^n*((X : ZPolynomial)^n-1) := by rw [geometric_mul]
        _ = signed ((monomial n c).comp (X^2))-signed (monomial n c) := by
          simp only [signed,Polynomial.map_comp,Polynomial.map_monomial,
            Polynomial.map_pow,Polynomial.map_X,Polynomial.monomial_comp]
          change C (c : Int)*X^n*((X : ZPolynomial)^n-1) =
            C (c : Int)*((X : ZPolynomial)^2)^n-monomial n (c : Int)
          rw [hp,←Polynomial.C_mul_X_pow_eq_monomial]
          ring

/-- Nonnegative coefficients make equal values at two distinct positive
points possible only for a constant polynomial. No degree bound is assumed. -/
theorem two_point_constant (p : NPolynomial) (x y : Real) (hx : 0 < x) (hxy : x < y)
    (he : p.eval₂ (Nat.castRingHom Real) x = p.eval₂ (Nat.castRingHom Real) y) :
    p = C (p.coeff 0) := by
  ext n
  by_cases hn : n=0
  · subst n
    simp
  · suffices hz : p.coeff n=0 by simpa [Polynomial.coeff_C,hn] using hz
    by_contra hc
    have hmem : n ∈ p.support := Polynomial.mem_support_iff.mpr hc
    have hpos : (0 : Real) < p.coeff n := by exact_mod_cast (Nat.pos_of_ne_zero hc)
    have hle : ∀ i ∈ p.support,
        (p.coeff i : Real)*x^i ≤ (p.coeff i : Real)*y^i := by
      intro i hi
      exact mul_le_mul_of_nonneg_left (pow_le_pow_left₀ hx.le hxy.le i) (by exact_mod_cast Nat.zero_le (p.coeff i))
    have hlt := Finset.sum_lt_sum hle
      ⟨n,hmem,mul_lt_mul_of_pos_left (pow_lt_pow_left₀ hxy hx.le hn) hpos⟩
    have hvalue : p.eval₂ (Nat.castRingHom Real) x < p.eval₂ (Nat.castRingHom Real) y := by
      simpa only [Polynomial.eval₂_eq_sum,Polynomial.sum,Nat.coe_castRingHom] using hlt
    exact (ne_of_lt hvalue) he
