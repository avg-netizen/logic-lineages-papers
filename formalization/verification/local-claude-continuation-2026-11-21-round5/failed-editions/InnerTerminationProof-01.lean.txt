variable {O ι : Type} {r : Nat}

/-! Arithmetic over abstract naturals. -/

theorem pow_step (p p1 pb : Nat) (hp : 0 < p) (h3 : p1 = p * 3) (hle : p1 ≤ pb) :
    2 * p < pb := by omega

theorem two_pow_lt {a b : Nat} (h : a < b) : 2 * 3 ^ a < 3 ^ b :=
  pow_step _ _ _ (Nat.pow_pos (by decide)) Nat.pow_succ (Nat.pow_le_pow_right (by decide) h)

theorem charge_bounds (M M' V V' t A B B' : Nat) (hM' : M' ≤ A + B') (hM : M = A + B)
    (hV : V' + t = V) (hle : B' ≤ B) : M' ≤ M ∧ V' ≤ V := by omega

theorem charge_strict (M M' V V' t A B B' : Nat) (hM' : M' ≤ A + B') (hM : M = A + B)
    (hV : V' + t = V) (hle : B' ≤ B) (h : 0 < t ∨ B' < B) : M' + V' < M + V := by omega

theorem merged_strict (p n np B B' : Nat) (hp : 0 < p) (hn : 2 * p ≤ np) (hB' : B' ≤ p)
    (hB : B = np) : B' < B := by omega

/-! Finite sums. -/

theorem sum_union_le_nat {α : Type} [DecidableEq α] (s t : Finset α) (f : α → Nat) :
    ∑ x ∈ s ∪ t, f x ≤ ∑ x ∈ s, f x + ∑ x ∈ t, f x := by
  have h := Finset.sum_union_inter (s₁ := s) (s₂ := t) (f := f)
  omega

theorem sum_biUnion_le_nat {α γ : Type} [DecidableEq α] (S : Finset γ) (t : γ → Finset α)
    (f : α → Nat) : ∑ x ∈ S.biUnion t, f x ≤ ∑ c ∈ S, ∑ x ∈ t c, f x := by
  classical
  induction S using Finset.induction_on with
  | empty => simp
  | insert a s ha ih =>
    rw [Finset.biUnion_insert, Finset.sum_insert ha]
    exact (sum_union_le_nat _ _ _).trans (Nat.add_le_add_left ih _)

theorem card_pair_le {α : Type} [DecidableEq α] (a b : α) :
    ({a} ∪ {b} : Finset α).card ≤ 2 := by
  have h := Finset.card_union_le ({a} : Finset α) {b}
  rw [Finset.card_singleton, Finset.card_singleton] at h
  omega

/-- Fields inside one input column weigh at most the column weight each. -/
theorem column_weight (s t : Finset (ι × Fin r)) (x0 : Fin r) (hs : s ⊆ t)
    (ht : ∀ e ∈ t, e.2 = x0) :
    ∑ e ∈ s, 3 ^ (e.2 : Nat) ≤ t.card * 3 ^ (x0 : Nat) :=
  calc ∑ e ∈ s, 3 ^ (e.2 : Nat) ≤ s.card • 3 ^ (x0 : Nat) :=
        Finset.sum_le_card_nsmul _ _ _ (fun e he => le_of_eq (by rw [ht e (hs he)]))
    _ ≤ t.card * 3 ^ (x0 : Nat) := by
        rw [smul_eq_mul]
        exact Nat.mul_le_mul_right _ (Finset.card_le_card hs)

theorem singleton_column (a : ι) (x0 : Fin r) :
    ∀ e ∈ ({(a, x0)} : Finset (ι × Fin r)), e.2 = x0 := by
  intro e he
  have h := Finset.mem_singleton.mp he
  subst h
  rfl

theorem pair_column [DecidableEq ι] (a b : ι) (x0 : Fin r) :
    ∀ e ∈ ({(a, x0)} ∪ {(b, x0)} : Finset (ι × Fin r)), e.2 = x0 := by
  intro e he
  rcases Finset.mem_union.mp he with h | h
  · exact singleton_column a x0 e h
  · exact singleton_column b x0 e h

/-! One class occurrence. -/

theorem lowerTerm_support (x0 : Fin r) (q : ι) (a : ℤ) (lower : Option ι) :
    ∃ b : ι, (lowerTerm x0 a lower).support ⊆ {(b, x0)} := by
  cases lower with
  | none =>
    refine ⟨q, ?_⟩
    have h0 : (lowerTerm x0 a (none : Option ι)).support = ∅ := Finsupp.support_zero
    rw [h0]
    exact Finset.empty_subset _
  | some h => exact ⟨h, Finsupp.support_single_subset⟩

/-- The replacement of one class occurrence never weighs more than the
occurrence, and weighs strictly less unless its prefix survives. -/
theorem member_weight [DecidableEq ι] (x0 : Fin r) (q : ι) (a : ℤ) (c : ι × Fin r)
    (d : Disposition ι) (h : d.Admissible x0 c.2) :
    ∑ e ∈ (d.term x0 q a).support, 3 ^ (e.2 : Nat) ≤ 3 ^ (c.2 : Nat) ∧
    ((∀ s, d ≠ .survives s) →
      ∑ e ∈ (d.term x0 q a).support, 3 ^ (e.2 : Nat) < 3 ^ (c.2 : Nat)) := by
  cases d with
  | ended k =>
    have h0 : ((Disposition.ended k).term x0 q a).support = ∅ := Finsupp.support_zero
    rw [h0, Finset.sum_empty]
    exact ⟨Nat.zero_le _, fun _ => Nat.pow_pos (by decide)⟩
  | survives k =>
    have hx : c.2 = x0 := h
    have hb := column_weight ((Disposition.survives k).term x0 q a).support {(q, x0)} x0
      Finsupp.support_single_subset (singleton_column q x0)
    rw [Finset.card_singleton, Nat.one_mul] at hb
    rw [hx]
    exact ⟨hb, fun hns => absurd rfl (hns k)⟩
  | transferred lower k =>
    have hx : x0 < c.2 := h
    obtain ⟨b, hb⟩ := lowerTerm_support x0 q a lower
    have hsub : ((Disposition.transferred lower k).term x0 q a).support ⊆
        {(q, x0)} ∪ {(b, x0)} :=
      Finsupp.support_sub.trans (Finset.union_subset_union Finsupp.support_single_subset hb)
    have hw := (column_weight _ _ x0 hsub (pair_column q b x0)).trans
      (Nat.mul_le_mul_right _ (card_pair_le (q, x0) (b, x0)))
    have hlt := two_pow_lt (Fin.lt_def.mp hx)
    exact ⟨(hw.trans_lt hlt).le, fun _ => hw.trans_lt hlt⟩

/-- A class whose prefixes all survive is replaced inside one field. -/
theorem survivor_term_support (x0 : Fin r) (q : ι) (a : ℤ) (d : Disposition ι)
    (hd : ∃ s, d = .survives s) : (d.term x0 q a).support ⊆ {(q, x0)} := by
  obtain ⟨s, rfl⟩ := hd
  exact Finsupp.support_single_subset

theorem survivor_input (x0 x : Fin r) (d : Disposition ι) (hd : ∃ s, d = .survives s)
    (h : d.Admissible x0 x) : x = x0 := by
  obtain ⟨s, rfl⟩ := hd
  exact h

/-! The lemma. -/

theorem fullInnerTermination : FullInnerTermination := by
  intro O ι _ _ r R R' hrep
  obtain ⟨T, S, x0, q, d, hT, hS, hne, hzero, hadm, hval, hcnt, -⟩ := hrep
  -- ordinary values
  have hV : valueCount R' + T.card = valueCount R := by
    unfold valueCount
    rw [hval, Finsupp.support_filter]
    have hsd : ({x ∈ R.value.support | x ∉ T} : Finset O) = R.value.support \ T := by
      ext x
      simp only [Finset.mem_filter, Finset.mem_sdiff]
    rw [hsd]
    exact Finset.card_sdiff_add_card_eq_card hT
  -- counts
  have hsupp : R'.count.support ⊆
      (R.count.support \ S) ∪ S.biUnion fun c => ((d c).term x0 q (R.count c)).support := by
    rw [hcnt]
    refine Finsupp.support_add.trans (Finset.union_subset_union ?_ Finsupp.support_finsetSum)
    intro e he
    rw [Finsupp.support_filter, Finset.mem_filter] at he
    exact Finset.mem_sdiff.mpr he
  have hM' : countCharge R' ≤ ∑ e ∈ R.count.support \ S, 3 ^ (e.2 : Nat) +
      ∑ e ∈ S.biUnion fun c => ((d c).term x0 q (R.count c)).support, 3 ^ (e.2 : Nat) :=
    (Finset.sum_le_sum_of_subset hsupp).trans (sum_union_le_nat _ _ _)
  have hM : countCharge R = ∑ e ∈ R.count.support \ S, 3 ^ (e.2 : Nat) +
      ∑ c ∈ S, 3 ^ (c.2 : Nat) := (Finset.sum_sdiff hS).symm
  have hmem : ∑ c ∈ S, ∑ e ∈ ((d c).term x0 q (R.count c)).support, 3 ^ (e.2 : Nat) ≤
      ∑ c ∈ S, 3 ^ (c.2 : Nat) :=
    Finset.sum_le_sum fun c hc => (member_weight x0 q (R.count c) c (d c) (hadm c hc)).1
  have hle : ∑ e ∈ S.biUnion fun c => ((d c).term x0 q (R.count c)).support, 3 ^ (e.2 : Nat) ≤
      ∑ c ∈ S, 3 ^ (c.2 : Nat) := (sum_biUnion_le_nat _ _ _).trans hmem
  obtain ⟨h1, h2⟩ := charge_bounds _ _ _ _ _ _ _ _ hM' hM hV hle
  refine ⟨h1, h2, charge_strict _ _ _ _ _ _ _ _ hM' hM hV hle ?_⟩
  by_cases hT0 : 0 < T.card
  · exact Or.inl hT0
  right
  have hTe : T = ∅ := Finset.card_eq_zero.mp (Nat.eq_zero_of_not_pos hT0)
  have hSne : S.Nonempty := by
    rcases hne with h | h
    · rw [hTe] at h
      exact absurd h Finset.not_nonempty_empty
    · exact h
  rw [hTe, Finset.sum_empty, mul_zero, zero_add] at hzero
  have hcard : 2 ≤ S.card := by
    by_contra hlt
    have hpos := Finset.card_pos.mpr hSne
    have hone : S.card = 1 := by omega
    obtain ⟨c, hc⟩ := Finset.card_eq_one.mp hone
    have hcS : c ∈ S := by
      rw [hc]
      exact Finset.mem_singleton_self c
    rw [hc, Finset.sum_singleton] at hzero
    exact Finsupp.mem_support_iff.mp (hS hcS) hzero
  by_cases hall : ∀ c ∈ S, ∃ s, d c = .survives s
  · -- every prefix survives: at least two occurrences merge into one field
    have hU : (S.biUnion fun c => ((d c).term x0 q (R.count c)).support) ⊆ {(q, x0)} :=
      Finset.biUnion_subset.mpr fun c hc =>
        survivor_term_support x0 q (R.count c) (d c) (hall c hc)
    have hB' := column_weight _ _ x0 hU (singleton_column q x0)
    rw [Finset.card_singleton, Nat.one_mul] at hB'
    have hB : ∑ c ∈ S, 3 ^ (c.2 : Nat) = S.card * 3 ^ (x0 : Nat) := by
      rw [Finset.sum_congr rfl fun c hc => by
        rw [survivor_input x0 c.2 (d c) (hall c hc) (hadm c hc)]]
      rw [Finset.sum_const, smul_eq_mul]
    exact merged_strict _ S.card _ _ _ (Nat.pow_pos (by decide))
      (Nat.mul_le_mul_right _ hcard) hB' hB
  · -- some prefix is ended or transferred
    obtain ⟨c, hc, hns⟩ : ∃ c ∈ S, ∀ s, d c ≠ .survives s := by
      by_contra hcon
      apply hall
      intro c hc
      by_contra hx
      exact hcon ⟨c, hc, fun s hs => hx ⟨s, hs⟩⟩
    exact (sum_biUnion_le_nat _ _ _).trans_lt
      (Finset.sum_lt_sum
        (fun c hc => (member_weight x0 q (R.count c) c (d c) (hadm c hc)).1)
        ⟨c, hc, (member_weight x0 q (R.count c) c (d c) (hadm c hc)).2 hns⟩)

end LogicLineages.PermittedControl.InnerTerminationProof
