/-- One installed window per closed source universe on which the type split reduces to the
fresh application and the all-terms sentence to the type split and typed computability. -/
theorem exact_InstalledAccessibilityReduction : InstalledAccessibilityReduction := by
  intro S K closed
  obtain ⟨A, _, _, _, _, _, _, _, _, _, _, _, _, _, _, _, _, scalar, core, guard, rules⟩ :=
    exact_InstalledLambdaWindowWithScalarAtOriginalWindow S K closed
  refine ⟨literalProbeWindow A, fun c => ⟨?_, ?_⟩⟩
  · intro total
    have output : ApplicationResultCarrier A c := by
      intro α Γ x u t app
      rw [guard] at app
      exact ((rules c).applicationOutput α Γ x u t app).1
    exact exact_TypingCasesFromReports (literalProbeWindow A) c
      (exact_StackedLiteralProbeReports S K A _ scalar core c output) total
  · intro typing computed
    let B := literalProbeWindow A
    have arithmetic := (rules c).arithmetic
    have forward := (rules c).forward
    rw [← guard] at arithmetic forward
    let step := hop A.contextGuard A.typeGuard
      (rawWithTarget A.scalar A.contextGuard A.typeGuard A.raw A.least) A.least
    let q : OTerm (Empty ⊕ Fin 1) := .var (.inr 0)
    let body := sourceBody B.presentation c
    let domain := carrier B.presentation false c q
    let goal := membership false q
    have domainHyp : Nonempty (Derivation body [domain] domain) :=
      ⟨Derivation.hypothesis List.mem_cons_self⟩
    have result : Nonempty (Derivation body [domain] goal) := by
      apply typing (Empty ⊕ Fin 1) [domain] q goal domainHyp
      intro τ hτ
      let typed := typedAt (chosenH B.probes (sourceWindow K)) (formulaType τ) q
      let Γ := [typed, domain]
      have typedHyp : Nonempty (Derivation body Γ typed) :=
        ⟨Derivation.hypothesis List.mem_cons_self⟩
      have heldDomain : Nonempty (Derivation body Γ domain) :=
        ⟨Derivation.hypothesis (List.mem_cons_of_mem _ List.mem_cons_self)⟩
      exact exact_TupleComputabilityToIFromArithmetic S.entries (fixed S K) (sourceWindow K)
        B.scalar (jRaw B.scalar) B.path (dCore B.scalar B.path permutation) B.probes c
        step neutralSource bottomSource B.installed.1 arithmetic forward
        (formulaType τ) (Empty ⊕ Fin 1) Γ q typedHyp (computed τ hτ _ Γ q heldDomain typedHyp)
    obtain ⟨proof⟩ := result
    exact ⟨Derivation.universalIntroduction (Derivation.implicationIntroduction proof)⟩
