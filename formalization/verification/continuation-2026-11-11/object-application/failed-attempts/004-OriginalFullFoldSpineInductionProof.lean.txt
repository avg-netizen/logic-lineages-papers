variable {body : Positive (Fin 1)} {alpha : Type} {Gamma : List (Formula alpha)}

/-- The original opening guard either exposes a strictly smaller function
payload, or supplies the actual selected leaf of the SAME full fold. -/
theorem stepAtInput (base : Data) (window : Program 1)
    (q input typ out parameter mark : OrdinaryArithmetic.Term alpha)
    (hb : Nonempty (Derivation body Gamma
      (belowPayload (SpineApplicationTargets.selectedData base window) window q)))
    (hr : Nonempty (Derivation body Gamma (graph1 InternalUnpair.right q input)))
    (hp : Nonempty (Derivation body Gamma
      (premises (SpineApplicationTargets.selectedData base window) window out input typ parameter))) :
    Nonempty (Derivation body Gamma (conclusion out input mark)) := by
  let data := SpineApplicationTargets.selectedData base window
  let accepted := graph2 (InternalParameterizedApplicationResultStack.openGuardP
    (InternalParameterizedCanonicalResultFold.rootGuard data)) (suc z) input parameter
  obtain ⟨hb⟩ := hb
  obtain ⟨hr⟩ := hr
  obtain ⟨hp⟩ := hp
  obtain ⟨hy⟩ := OriginalFullFoldSpineOpeningProof.acceptedAtOutput
    (body := body) (Gamma := accepted :: Gamma) base window q input typ out parameter mark
    ⟨addHyp hb accepted⟩ ⟨addHyp hr accepted⟩ ⟨addHyp hp accepted⟩
    ⟨Derivation.hypothesis List.mem_cons_self⟩
  have hz : Nonempty (Derivation body (accepted.not :: Gamma)
      (graph2 (InternalParameterizedApplicationResultStack.openGuardP
        (InternalParameterizedCanonicalResultFold.rootGuard data)) z input parameter)) :=
    InternalBoundedDescentNormalizerStepSupportProof.conjunctionReject _ _
      (Fin.cons input (Fin.cons parameter Fin.elim0)) ⟨Derivation.hypothesis List.mem_cons_self⟩
  have hpn : Nonempty (Derivation body (accepted.not :: Gamma)
      (premises data window out input typ parameter)) := ⟨addHyp hp accepted.not⟩
  obtain ⟨hn⟩ := OriginalFullFoldSpineLeafProof.rejectedMarkIff
    (body := body) (Gamma := accepted.not :: Gamma) base window out input parameter mark
    (InternalApplicationStepPeelSupportProof.andLeft hpn)
    (InternalApplicationStepPeelSupportProof.andLeft
      (InternalApplicationStepPeelSupportProof.andLeft (InternalApplicationStepPeelSupportProof.andRight hpn)))
    (InternalApplicationStepPeelSupportProof.andRight (InternalApplicationStepPeelSupportProof.andRight hpn)) hz
  exact ⟨disjunctionElimination (InternalControlGraphChoiceProof.excludedMiddle accepted)
    (Derivation.implicationIntroduction hy) (Derivation.implicationIntroduction hn)⟩

/-- Close all five variables inside the original arithmetic induction row. -/
theorem payloadStep (base : Data) (window : Program 1) (q : OrdinaryArithmetic.Term alpha)
    (hb : Nonempty (Derivation body Gamma
      (belowPayload (SpineApplicationTargets.selectedData base window) window q))) :
    Nonempty (Derivation body Gamma
      (atPayload (SpineApplicationTargets.selectedData base window) window q)) := by
  let data := SpineApplicationTargets.selectedData base window
  apply atPayloadIntro data window q
  let input : OrdinaryArithmetic.Term (alpha ⊕ Fin 5) := .var (.inr 0)
  let typ : OrdinaryArithmetic.Term (alpha ⊕ Fin 5) := .var (.inr 1)
  let out : OrdinaryArithmetic.Term (alpha ⊕ Fin 5) := .var (.inr 2)
  let parameter : OrdinaryArithmetic.Term (alpha ⊕ Fin 5) := .var (.inr 3)
  let mark : OrdinaryArithmetic.Term (alpha ⊕ Fin 5) := .var (.inr 4)
  let Delta := premises data window out input typ parameter ::
    graph1 InternalUnpair.right (q.relabel Sum.inl) input :: Gamma.map (fun p => p.relabel Sum.inl)
  have hb' := renameInto hb (Sum.inl : alpha → alpha ⊕ Fin 5) Delta (by
    intro p hp
    exact List.mem_cons_of_mem _ (List.mem_cons_of_mem _ (List.mem_map.mpr ⟨p,hp,rfl⟩)))
  rw [belowPayloadRename] at hb'
  obtain ⟨h⟩ := stepAtInput base window (q.relabel Sum.inl) input typ out parameter mark hb'
    (show Nonempty (Derivation body Delta (graph1 InternalUnpair.right (q.relabel Sum.inl) input)) from
      ⟨Derivation.hypothesis (List.mem_cons_of_mem _ List.mem_cons_self)⟩)
    (show Nonempty (Derivation body Delta (premises data window out input typ parameter)) from
      ⟨Derivation.hypothesis List.mem_cons_self⟩)
  exact ⟨Derivation.implicationIntroduction (Derivation.implicationIntroduction h)⟩

/-- Strong induction discharges the smaller-payload premise internally. -/
theorem payloadSpine (base : Data) (window : Program 1) (payload : OrdinaryArithmetic.Term alpha) :
    Nonempty (Derivation body Gamma
      (atPayload (SpineApplicationTargets.selectedData base window) window payload)) := by
  apply payloadInduction (SpineApplicationTargets.selectedData base window) window payload
  intro beta Delta q hb
  exact payloadStep base window q hb
