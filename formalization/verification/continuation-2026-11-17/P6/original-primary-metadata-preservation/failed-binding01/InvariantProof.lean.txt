variable {body : Positive (Fin 1)} {alpha : Type} {Delta : List (Formula alpha)}

theorem eliminate (n : Nat) (arity : Fin n → Nat)
    (reached candidate typ raw rootTypeRef boundsRef : OrdinaryArithmetic.Term alpha)
    (hi : Nonempty (Derivation body Delta (allAccepted n arity reached rootTypeRef boundsRef)))
    (hm : Nonempty (Derivation body Delta (graph2 contains (numeral 1) reached candidate)))
    (: Nonempty (Derivation body Delta (graph2 InternalPairing.pairing candidate typ raw))) :
    Nonempty (Derivation body Delta (accepted n arity typ raw rootTypeRef boundsRef)) := by
  obtain ⟨h⟩ := elimination body alpha 3 Delta _ (fun i => emb (![candidate,typ,raw] i)) hi
  rw [memberWitness] at h
  obtain ⟨ha⟩ := andIntro hm exact ⟨Derivation.implicationElimination h ha⟩

theorem introduce (n : Nat) (arity : Fin n → Nat)
    (reached rootTypeRef boundsRef : OrdinaryArithmetic.Term alpha)
    (next : ∀ (beta : Type) (j : alpha → beta) (Theta : List (Formula beta))
      (candidate typ raw : OrdinaryArithmetic.Term beta),
      (∀ q ∈ Delta, q.relabel j ∈ Theta) →
      Nonempty (Derivation body Theta
        (graph2 contains (numeral 1) (reached.relabel j) candidate)) →
      Nonempty (Derivation body Theta (graph2 InternalPairing.pairing candidate typ raw)) →
      Nonempty (Derivation body Theta
        (accepted n arity typ raw (rootTypeRef.relabel j) (boundsRef.relabel j)))) :
    Nonempty (Derivation body Delta (allAccepted n arity reached rootTypeRef boundsRef)) := by
  apply introduction body alpha 3 Delta
  let candidate : OrdinaryArithmetic.Term (alpha ⊕ Fin 3) := .var (.inr 0)
  let typ : OrdinaryArithmetic.Term (alpha ⊕ Fin 3) := .var (.inr 1)
  let raw : OrdinaryArithmetic.Term (alpha ⊕ Fin 3) := .var (.inr 2)
  let A := graph2 contains (numeral 1) (reached.relabel Sum.inl) candidate ⊓
    graph2 InternalPairing.pairing candidate typ raw
  obtain ⟨h⟩ := next (alpha ⊕ Fin 3) Sum.inl
    (A :: Delta.map (fun q => q.relabel Sum.inl)) candidate typ raw
    (fun q hq => List.mem_cons_of_mem _ (List.mem_map.mpr ⟨q,hq,rfl⟩))
    ⟨Derivation.conjunctionLeft (Derivation.hypothesis List.mem_cons_self)⟩
    ⟨Derivation.conjunctionRight (Derivation.hypothesis List.mem_cons_self)⟩
  exact ⟨Derivation.implicationIntroduction h⟩
