theorem graphOneSubst {alpha beta : Type} (P : Program 1) (out arg : OTerm alpha) (ts : alpha → OTerm beta) :
    (graph1 P out arg).subst (fun a => emb (ts a)) = graph1 P (out.subst ts) (arg.subst ts) := by
  simp only [graph1,programSubst,consTermSubst,emptySubst]
def fieldsSchema {alpha : Type} (weight : Program 1) (left right ls rs : OTerm alpha) : Formula alpha :=
  agreement left right ⊓ (graph1 weight ls left ⊓ graph1 weight rs right)
def matrixSchema {alpha : Type} (weight : Program 1) (left : OTerm alpha) : Formula (alpha ⊕ Fin 3) :=
  let x : Fin 3 → OTerm (alpha ⊕ Fin 3) := fun i => .var (.inr i)
  (fieldsSchema weight (left.relabel Sum.inl) (x 0) (x 1) (x 2)).imp (eqn (x 1) (x 2))
def allSchema {alpha : Type} (weight : Program 1) (raw : OTerm alpha) : Formula alpha :=
  InternalClosedPrograms.allBlock (matrixSchema weight raw)
theorem fieldsSchemaSubst {alpha beta : Type} (weight : Program 1) (left right ls rs : OTerm alpha) (ts : alpha → OTerm beta) :
    (fieldsSchema weight left right ls rs).subst (fun a => emb (ts a)) =
      fieldsSchema weight (left.subst ts) (right.subst ts) (ls.subst ts) (rs.subst ts) := by
  simp only [fieldsSchema,conjunctionSubst,agreementSubst,graphOneSubst]
theorem allSchemaSubst {alpha beta : Type} (weight : Program 1) (raw : OTerm alpha) (ts : alpha → OTerm beta) :
    (allSchema weight raw).subst (fun a => emb (ts a)) = allSchema weight (raw.subst ts) := by
  unfold allSchema matrixSchema
  rw [allBlockSubst,liftedEmbedding]
  simp only [impSubst,fieldsSchemaSubst,eqnSubst,liftSubst,Term.subst,Sum.elim_inr]
theorem matrixSchemaWitness {alpha : Type} (weight : Program 1) (left right ls rs : OTerm alpha) :
    (matrixSchema weight left).subst (Sum.elim Term.var (fun i => emb (![right,ls,rs] i))) =
      (fieldsSchema weight left right ls rs).imp (eqn ls rs) := by
  have he : (Sum.elim Term.var (fun i => emb (![right,ls,rs] i))) =
      (fun a => emb (Sum.elim Term.var ![right,ls,rs] a)) := by
    funext a; cases a <;> rfl
  unfold matrixSchema
  rw [he]
  simp only [impSubst,fieldsSchemaSubst,eqnSubst,witnessLift,Term.subst,Sum.elim_inr]
  rfl
