theorem graphOneSubst {alpha beta : Type} (P : Program 1) (out x : OTerm alpha) (ts : alpha → OTerm beta) :
    (graph1 P out x).subst (fun a => emb (ts a)) = graph1 P (out.subst ts) (x.subst ts) := by
  simp only [graph1,programSubst,consTermSubst,emptySubst]
theorem graphTwoSubst {alpha beta : Type} (P : Program 2) (out x y : OTerm alpha) (ts : alpha → OTerm beta) :
    (graph2 P out x y).subst (fun a => emb (ts a)) = graph2 P (out.subst ts) (x.subst ts) (y.subst ts) := by
  simp only [graph2,programSubst,consTermSubst,emptySubst]

theorem fieldsSubst {alpha beta : Type} (n : Nat) (arity : Fin n → Nat)
    (source target node window lv rv : OTerm alpha) (ts : alpha → OTerm beta) :
    (fields n arity source target node window lv rv).subst (fun a => emb (ts a)) =
      fields n arity (source.subst ts) (target.subst ts) (node.subst ts) (window.subst ts) (lv.subst ts) (rv.subst ts) := by
  simp only [fields,conjunctionSubst,graphOneSubst,graphTwoSubst,sucSubst,zSubst]
@[simp] theorem fieldsRename {alpha beta : Type} (n : Nat) (arity : Fin n → Nat)
    (source target node window lv rv : OTerm alpha) (j : alpha → beta) :
    (fields n arity source target node window lv rv).relabel j =
      fields n arity (source.relabel j) (target.relabel j) (node.relabel j) (window.relabel j) (lv.relabel j) (rv.relabel j) := by
  rw [formulaRelabelAsSubst]
  change (fields n arity source target node window lv rv).subst (fun a => emb (.var (j a))) = _
  rw [fieldsSubst]
  simp only [← termRelabelAsSubst]

theorem atRawSubst {alpha beta : Type} (n : Nat) (arity : Fin n → Nat) (raw : OTerm alpha) (ts : alpha → OTerm beta) :
    (AtRaw n arity raw).subst (fun a => emb (ts a)) = AtRaw n arity (raw.subst ts) := by
  unfold AtRaw matrix
  rw [allBlockSubst,liftedEmbedding]
  simp only [impSubst,fieldsSubst,eqnSubst,liftSubst,Term.subst,Sum.elim_inr]
@[simp] theorem atRawRename {alpha beta : Type} (n : Nat) (arity : Fin n → Nat) (raw : OTerm alpha) (j : alpha → beta) :
    (AtRaw n arity raw).relabel j = AtRaw n arity (raw.relabel j) := by
  rw [formulaRelabelAsSubst]
  change (AtRaw n arity raw).subst (fun a => emb (.var (j a))) = _
  rw [atRawSubst]
  simp only [← termRelabelAsSubst]

theorem matrixWitness {alpha : Type} (n : Nat) (arity : Fin n → Nat) (source target node window lv rv : OTerm alpha) :
    (matrix n arity source).subst (Sum.elim Term.var (fun i => emb (![target,node,window,lv,rv] i))) =
      (fields n arity source target node window lv rv).imp (eqn lv rv) := by
  have he : (Sum.elim Term.var (fun i => emb (![target,node,window,lv,rv] i))) =
      (fun a => emb (Sum.elim Term.var ![target,node,window,lv,rv] a)) := by
    funext a; cases a <;> rfl
  unfold matrix
  rw [he]
  simp only [impSubst,fieldsSubst,eqnSubst,witnessLift,Term.subst,Sum.elim_inr]
  rfl
theorem predicateInstantiate {alpha : Type} (n : Nat) (arity : Fin n → Nat) (raw : OTerm alpha) :
    instantiate (predicate n arity) (emb raw) = AtRaw n arity raw := by
  rw [predicate,instantiate,bindEmbedding,atRawSubst]
  simp only [varBind]

@[simp] theorem belowRawRename {alpha beta : Type} (n : Nat) (arity : Fin n → Nat) (raw : OTerm alpha) (j : alpha → beta) :
    (BelowRaw n arity raw).relabel j = BelowRaw n arity (raw.relabel j) := by
  unfold BelowRaw
  rw [InternalStrongInductionSyntaxProof.belowRename]
  unfold predicate
  rw [atRawRename]
  rfl

theorem predicateLift {alpha : Type} (n : Nat) (arity : Fin n → Nat) :
    lift (predicate (alpha := alpha) n arity) = predicate (alpha := alpha ⊕ Fin 1) n arity := by
  unfold lift predicate
  rw [atRawRename]
