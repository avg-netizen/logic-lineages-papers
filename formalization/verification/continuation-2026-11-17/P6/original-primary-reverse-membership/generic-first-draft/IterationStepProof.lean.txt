theorem successorCore (E : Program 3) (count pool reached parameter out carrier seed : OTerm alpha)
 (hback : Nonempty (Derivation body Delta (backClosed E pool carrier parameter)))
 (hrow : Nonempty (Derivation body Delta (AtCount E count pool reached parameter carrier seed)))
 (hi : Nonempty (Derivation body Delta (graph4 (PrimaryIteration.iterate E) out (suc count) pool reached parameter)))
 (hall : Nonempty (Derivation body Delta (allAccepted pool carrier seed reached))) :
 Nonempty (Derivation body Delta (allAccepted pool carrier seed out)) := by
 apply programCases (PrimaryIteration.iterate E) (Fin.cons count (Fin.cons pool (Fin.cons reached (Fin.cons parameter Fin.elim0)))) _
 intro beta j Theta previous hj hr
 rw [allAcceptedRename]
 simp only [consTermRename,emptyRename] at hr
 have hback' := renameInto hback j Theta hj
 rw [backClosedRename] at hback'
 have hrow' := renameInto hrow j Theta hj
 rw [atCountRename] at hrow'
 have hi' := renameInto hi j Theta hj
 simp only [graph4Rename,sucRelabel] at hi'
 have hall' := renameInto hall j Theta hj
 rw [allAcceptedRename] at hall'
 have hexpand := PrimaryIteration.RecursionProof.successorRecover E (count.relabel j) previous (out.relabel j)
   (pool.relabel j) (reached.relabel j) (parameter.relabel j) hr hi'
 have hprevious := rowApply E (count.relabel j) (pool.relabel j) (reached.relabel j) (parameter.relabel j)
   previous (carrier.relabel j) (seed.relabel j) hrow' hr hall'
 exact ExpansionProof.preserved E (pool.relabel j) (carrier.relabel j) (seed.relabel j)
   previous (parameter.relabel j) (out.relabel j) hexpand hprevious hback'
