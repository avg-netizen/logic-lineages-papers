theorem allCounts (E : Program 3) (count pool reached parameter carrier seed : OTerm alpha)
 (hback : Nonempty (Derivation body Delta (backClosed E pool carrier parameter))) :
 Nonempty (Derivation body Delta (AtCount E count pool reached parameter carrier seed)) := by
 let k : OTerm (alpha ⊕ Fin 1) := .var (.inr 0)
 let theta := AtCount E k (pool.relabel Sum.inl) (reached.relabel Sum.inl)
   (parameter.relabel Sum.inl) (carrier.relabel Sum.inl) (seed.relabel Sum.inl)
 have hz : Nonempty (Derivation body Delta (instantiate theta (emb z))) := by
  simpa only [theta,k,instantiate,bindEmbedding,atCountSubst,varBind,liftBind] using
   (rowZero (body := body) (Delta := Delta) E pool reached parameter carrier seed)
 have hs : Nonempty (Derivation body (theta :: Delta.map weaken) (successorInstance theta)) := by
  let D := theta :: Delta.map weaken
  have hj : ∀ p ∈ Delta, p.relabel (Sum.inl : alpha → alpha ⊕ Fin 1) ∈ D := by
   intro p hp; exact List.mem_cons_of_mem _ (List.mem_map.mpr ⟨p,hp,rfl⟩)
  have hback' := renameInto hback Sum.inl D hj
  rw [backClosedRename] at hback'
  have h := rowSuccessor E k (pool.relabel Sum.inl) (reached.relabel Sum.inl)
    (parameter.relabel Sum.inl) (carrier.relabel Sum.inl) (seed.relabel Sum.inl) hback' ⟨Derivation.hypothesis List.mem_cons_self⟩
  simpa only [theta,k,successorInstance,nextEmbedding,atCountSubst,nextTerm,Term.subst,Sum.elim_inr,nextLift] using h
 obtain ⟨hz⟩ := hz
 obtain ⟨hs⟩ := hs
 have h := Derivation.universalElimination (emb count) (natInduction theta hz hs)
 simpa only [theta,k,instantiate,bindEmbedding,atCountSubst,varBind,liftBind] using Nonempty.intro h
