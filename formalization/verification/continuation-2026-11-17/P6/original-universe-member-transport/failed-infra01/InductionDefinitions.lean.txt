def Transport {alpha : Type} (H : Program 1) (M N W : Program 2) (arguments : Bool)
    (typ raw gamma c : OTerm alpha) : Formula alpha :=
  (MemberIncluded M N c).imp
    ((Accepted H M W arguments typ raw gamma c).imp (Accepted H N W arguments typ raw gamma c))
def transportMatrix {alpha : Type} (H : Program 1) (M N W : Program 2) (raw : OTerm alpha) : Formula (alpha ⊕ Fin 3) :=
  let v : Fin 3 → OTerm (alpha ⊕ Fin 3) := fun i => .var (.inr i)
  Transport H M N W false (v 0) (raw.relabel Sum.inl) (v 1) (v 2) ⊓
    Transport H M N W true (v 0) (raw.relabel Sum.inl) (v 1) (v 2)
def AtRaw {alpha : Type} (H : Program 1) (M N W : Program 2) (raw : OTerm alpha) : Formula alpha :=
  InternalClosedPrograms.allBlock (transportMatrix H M N W raw)
def rawPredicate {alpha : Type} (H : Program 1) (M N W : Program 2) : Formula (alpha ⊕ Fin 1) :=
  AtRaw H M N W (.var (.inr 0))
def BelowRaw {alpha : Type} (H : Program 1) (M N W : Program 2) (raw : OTerm alpha) : Formula alpha :=
  InternalStrongInduction.below (rawPredicate H M N W) raw
